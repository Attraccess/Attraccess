cmake_minimum_required(VERSION 3.24)
project(attractap_lvgl_memory_tests LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 99)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
get_filename_component(FIRMWARE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LVGL_SOURCE_DIR "" CACHE PATH "Existing LVGL 9.3.0 source tree")
if(NOT LVGL_SOURCE_DIR AND EXISTS "${FIRMWARE_DIR}/managed_components/lvgl__lvgl/CMakeLists.txt")
    set(LVGL_SOURCE_DIR "${FIRMWARE_DIR}/managed_components/lvgl__lvgl")
endif()
if(LVGL_SOURCE_DIR)
    add_subdirectory("${LVGL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/lvgl" EXCLUDE_FROM_ALL)
else()
    include(FetchContent)
    FetchContent_Declare(lvgl
        URL https://codeload.github.com/lvgl/lvgl/tar.gz/108e5aff3c90cc4d969331ed61aff2bbd365d430
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(lvgl)
endif()
target_compile_definitions(lvgl PRIVATE ESP_PLATFORM LV_CONF_KCONFIG_EXTERNAL_INCLUDE="sdkconfig.h")
target_include_directories(lvgl PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/fakes")
add_executable(lvgl_memory_test lvgl_memory_test.c)
target_compile_definitions(lvgl_memory_test PRIVATE ESP_PLATFORM LV_CONF_KCONFIG_EXTERNAL_INCLUDE="sdkconfig.h")
target_include_directories(lvgl_memory_test PRIVATE fakes)
target_link_libraries(lvgl_memory_test PRIVATE lvgl m)
enable_testing()
add_test(NAME lvgl_memory_pool_in_psram COMMAND lvgl_memory_test)
set_tests_properties(lvgl_memory_pool_in_psram PROPERTIES TIMEOUT 20)
