cmake_minimum_required(VERSION 3.24)
project(attractap_websocket_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(FIRMWARE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(websocket-pong-probe pong_probe_test.cpp "${FIRMWARE_DIR}/src/websocket/pong_probe.cpp")
target_include_directories(websocket-pong-probe PRIVATE shims "${FIRMWARE_DIR}/src")
target_compile_definitions(websocket-pong-probe PRIVATE ESP_PLATFORM=1 LOGGER_LEVEL_NUM=0)
enable_testing()
add_test(NAME websocket-pong-probe COMMAND websocket-pong-probe)
set_tests_properties(websocket-pong-probe PROPERTIES TIMEOUT 5)
